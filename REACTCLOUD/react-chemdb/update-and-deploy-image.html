1. Update Source

git checkout blurock-reactcloudrun-setup
git pull --ff-only

2. Build a New Image Tag

export PROJECT_ID="blurock-database"
export REGION="us-central1"
export SERVICE_NAME="react-chemdb"
export REPO_NAME="react-images"
export TAG="v2-amd64"
export IMAGE_URI="$REGION-docker.pkg.dev/$PROJECT_ID/$REPO_NAME/$SERVICE_NAME:$TAG"

docker buildx build --platform linux/amd64 -t "$IMAGE_URI" --push .

3. Deploy to Cloud Run

gcloud run deploy "$SERVICE_NAME" \
  --image "$IMAGE_URI" \
  --platform managed \
  --region "$REGION" \
  --allow-unauthenticated \
  --memory 2Gi \
  --timeout 3600 \
  --cpu 2 \
  --concurrency 1 \
  --max-instances 10

4. Validate

SERVICE_URL="$(gcloud run services describe "$SERVICE_NAME" --region "$REGION" --format='value(status.url)')"
curl "$SERVICE_URL/health"
curl -X POST "$SERVICE_URL/api/run-input" \
  -H 'Content-Type: application/json' \
  -d '{"inputFile":"PrintRxnPatternsList.inp","root":"printpats"}'
Use a new immutable tag for every deploy.
